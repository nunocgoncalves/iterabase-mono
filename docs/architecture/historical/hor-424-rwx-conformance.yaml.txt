  name: worker-a

spec:
  suspend: true
  backoffLimit: 0
  template:
    spec:
      restartPolicy: Never
      initContainers:
        - name: establish-root
          image: debian:13-slim@sha256:d7e12182ce18b85b93007c1dedf31f2d29e01ccf3182cc4017c709b6259bc132
          command: [bash, -ceu]
          args:
            - |
              chown 0:0 /sessions
              chmod 0711 /sessions
              test "$(stat -c %u:%g:%a /sessions)" = 0:0:711
              test "$(stat -c %u:%g:%a /sessions/session-a)" = 10001:10001:700
              test "$(stat -c %u:%g:%a /sessions/session-b)" = 10002:10002:700
          securityContext:
            runAsUser: 0
            runAsGroup: 0
          volumeMounts:
            - name: sessions
              mountPath: /sessions
      containers:
        - name: worker
          image: debian:13-slim@sha256:d7e12182ce18b85b93007c1dedf31f2d29e01ccf3182cc4017c709b6259bc132
          command: [bash, -ceu]
          args:
            - |
              test "$(cat /sessions/session-a/secret)" = seed-a
              if ls /sessions >/dev/null 2>&1; then
                echo parent_listing=unexpected
                exit 10
              fi
              if cat /sessions/session-b/secret >/dev/null 2>&1; then
                echo sibling_read=unexpected
                exit 11
              fi
              if printf intrusion 2>/dev/null > /sessions/session-b/intrusion; then
                echo sibling_write=unexpected
                exit 12
              fi
              touch /sessions/.conformance/worker-a-ready
              for _ in $(seq 1 120); do
                test -f /sessions/.conformance/worker-b-ready && break
                sleep 1
              done
              test -f /sessions/.conformance/worker-b-ready

              printf atomic-a > /sessions/session-a/atomic.tmp
              sync -f /sessions/session-a/atomic.tmp
              mv /sessions/session-a/atomic.tmp /sessions/session-a/atomic.final
              sync -d /sessions/session-a
              test "$(cat /sessions/session-a/atomic.final)" = atomic-a
              printf remove-a > /sessions/session-a/remove-me
              rm /sessions/session-a/remove-me
              test ! -e /sessions/session-a/remove-me

              start=$(date +%s%N)
              dd if=/dev/zero of=/sessions/session-a/stream.bin bs=1M count=32 conv=fsync status=none
              end=$(date +%s%N)
              stream_ms=$(( (end-start)/1000000 ))
              start=$(date +%s%N)
              for i in $(seq 1 250); do
                dd if=/dev/zero of="/sessions/session-a/small-$i" bs=4096 count=1 status=none
              done
              sync
              end=$(date +%s%N)
              small_ms=$(( (end-start)/1000000 ))
              sha256sum /sessions/session-a/stream.bin | cut -d" " -f1 > /sessions/session-a/stream.sha256
              echo "worker=a isolation=pass atomic_rename=pass fsync=pass unlink=pass stream_bytes=33554432 stream_ms=$stream_ms small_files=250 small_bytes=1024000 small_ms=$small_ms"
          securityContext:
            runAsUser: 10001
            runAsGroup: 10001
            allowPrivilegeEscalation: false
          volumeMounts:
            - name: sessions
              mountPath: /sessions
      volumes:
        - name: sessions
          persistentVolumeClaim:
            claimName: sessions
